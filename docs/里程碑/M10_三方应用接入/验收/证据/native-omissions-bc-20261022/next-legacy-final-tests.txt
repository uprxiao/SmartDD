

describe('registered legacy H damage retains ordinary native work',()=>{
 beforeEach(()=>{
  // This private TestPtys composition never forks CLI/MCP children. Match each
  // synthetic bridge exit to its own synthetic CLI exit, as in the existing controls.
  const handles=new Map<string,Awaited<ReturnType<typeof managed.broker.prepare>>['bridge']>(),prepare=managed.broker.prepare.bind(managed.broker),kill=ptys.kill.bind(ptys);
  vi.spyOn(managed.broker,'prepare').mockImplementation(async(...args)=>{const value=await prepare(...args);handles.set(args[0].ptyId,value.bridge);return value;});
  vi.spyOn(ptys,'kill').mockImplementation(id=>{kill(id);const handle=handles.get(id);if(handle){handles.delete(id);void handle.exited();}});
 });
 it.each(['claude-code','codex'].flatMap(runner=>['new','resume'].flatMap(mode=>['owned-byte-edit','user-replacement'].map(fault=>[runner,mode,fault] as const))))('retains registered legacy H and full original C without application grants: %s %s %s',async(runner,mode,fault)=>{
  const old=capabilities.prepare({projectId,taskId:randomUUID(),runner:'codex',cwd,home,profileArgs:[]}),originalPath=old.snapshot.skills.find(s=>s.enabled)!.path;old.release();
  await applications.selections.setAdded('mail',true,0);await applications.selections.select({projectId,appId:'mail',enabled:true,expectedRevision:0,connections:{foxmail:null,smtp:null}});
  const h=await create(codexId),hId=await stopForSwitchFixture(h,'codex'),catalogFile=join(home,'.agents/smartdd-runtime/catalog.json'),catalog=readFileSync(catalogFile),raw=JSON.parse(catalog.toString()),entry=raw.entries.find((e:{path:string})=>e.path===originalPath);expect(entry).toMatchObject({format:'legacy-tree-v1',storeId:support.store.storeId});
  const profile=runner==='codex'?codexId:claudeId,seed=mode==='resume'?(runner==='codex'?h:await create(claudeId)):undefined,nativeId=seed?(runner==='codex'?hId:await stopForSwitchFixture(seed,'claude-code')):undefined,fixed=seed?await managed.sessions.get(seed.task.id):undefined,refs=seed?await managed.sessions.references(seed.task.id):[],hResource=support.store.get(entry.resourceId),sourceRow=sqlite.prepare('SELECT dependency_id AS id FROM capability_resource_dependencies WHERE resource_id=?').get(entry.resourceId) as {id:string},source=support.store.get(sourceRow.id),original=readFileSync(originalPath),folder=dirname(originalPath),backup=join(root,'BD-original-legacy-H');
  if(fault==='user-replacement'){renameSync(folder,backup);mkdirSync(folder);writeFileSync(originalPath,'---\nname: user-legacy-replacement\ndescription: preserve this user directory\n---\n');}else writeFileSync(originalPath,Buffer.concat([original,Buffer.from('\nEdited owned legacy bytes.\n')]));
  const user=readFileSync(originalPath),command=profiles.resolve(profile).command;writeFileSync(command,'#!/bin/sh\ncase "$1" in --version) echo "99.0.0";; *) exit 70;; esac\n');let running:Awaited<ReturnType<typeof create>>|undefined;
  try{running=seed?await tasks.open({taskId:seed.task.id}):await create(profile);const run=running.task.runtimeSnapshot!.runId;expect(ptys.live.has(running.ptyId)).toBe(true);expect(()=>managed.runs.assertActive(run)).toThrow('不签发');expect(managed.runs.inputContext(running.task.id,running.ptyId)).toBeNull();expect(sqlite.prepare("SELECT count(*) n FROM capability_run_processes WHERE run_id=? AND kind='bridge'").get(run)).toEqual({n:0});
   const held=sqlite.prepare("SELECT r.resource_id AS id,r.content_hash AS hash,r.generation FROM capability_lease_resources r JOIN capability_resource_leases l ON l.lease_id=r.lease_id WHERE l.owner_kind='run' AND l.owner_id=?").all(run);expect(held).toContainEqual({id:hResource.resourceId,hash:hResource.contentHash,generation:hResource.generation});expect(held).toContainEqual({id:source.resourceId,hash:source.contentHash,generation:source.generation});expect(readFileSync(catalogFile)).toEqual(catalog);expect(readFileSync(originalPath)).toEqual(user);
   const consumer=JSON.parse(readFileSync(join(home,'.agents/smartdd-runtime/catalog-consumers',support.store.storeId+'-'+run+'.json'),'utf8'));expect(consumer.generation).toBe(raw.generation);expect(consumer.runId).toBe(run);expect(consumer.ptyId).toBe(running.ptyId);expect(consumer.runner).toBe(runner);
   if(seed){expect((await managed.sessions.get(seed.task.id)).snapshot).toEqual(fixed!.snapshot);expect(running.task.nativeSessionId).toBe(nativeId);for(const ref of refs)expect(await managed.sessions.references(seed.task.id)).toContainEqual(ref);}expect(sqlite.prepare('SELECT count(*) n FROM mail_send_grants').get()).toEqual({n:0});
  }finally{if(running){tasks.kill(running.task.id);await vi.waitFor(()=>expect(managed.runs.get(running!.task.runtimeSnapshot!.runId).state).toBe('stopped'));}}
 },25000);
});
